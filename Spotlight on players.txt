local function removeESP(character)
    if espCache[character] then
        if espCache[character].highlight then espCache[character].highlight:Destroy() end
        if espCache[character].billboard then espCache[character].billboard:Destroy() end
        espCache[character] = nil
    end
end

local function addESP(character, modelName)
    if character == LocalPlayer.Character then return end
    if espCache[character] then return end
    if not character:FindFirstChild("HumanoidRootPart") then return end

    local highlight = Instance.new("Highlight")
    highlight.Name = "ScannerHighlight"
    highlight.Adornee = character
    highlight.FillColor = Color3.fromRGB(255, 0, 0)
    highlight.FillTransparency = 0.4
    highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
    highlight.OutlineTransparency = 0
    highlight.Parent = character

    local billboard = Instance.new("BillboardGui")
    billboard.Name = "ScannerBillboard"
    billboard.Adornee = character.HumanoidRootPart
    billboard.Size = UDim2.new(0, 200, 0, 60)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Parent = character

    local textLabel = Instance.new("TextLabel")
    textLabel.Size = UDim2.new(1, 0, 1, 0)
    textLabel.BackgroundTransparency = 1
    textLabel.TextColor3 = Color3.fromRGB(255, 50, 50)
    textLabel.TextStrokeTransparency = 0
    textLabel.TextSize = 13
    textLabel.Font = Enum.Font.GothamBold
    textLabel.Parent = billboard

    espCache[character] = {
        highlight = highlight,
        billboard = billboard,
        textLabel = textLabel,
        rootPart = character.HumanoidRootPart,
        name = modelName
    }
end

ScanButton.MouseButton1Click:Connect(function()
    scanningEnabled = not scanningEnabled
    
    if scanningEnabled then
        ScanButton.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
        for _, player in ipairs(Players:GetPlayers()) do
            if player.Character then
                addESP(player.Character, player.Name)
            end
        end
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("Model") and obj:FindFirstChild("Humanoid") and obj:FindFirstChild("HumanoidRootPart") then
                local isPlayerChar = false
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character == obj then isPlayerChar = true break end
                end
                if not isPlayerChar then
                    addESP(obj, obj.Name)
                end
            end
        end
    else
        ScanButton.BackgroundColor3 = Color3.fromRGB(35, 35, 40)
        for char, _ in pairs(espCache) do
            removeESP(char)
        end
        espCache = {}
    end
end)

RunService.RenderStepped:Connect(function()
    if activeSpectatingPlayer then
        if activeSpectatingPlayer.Character and activeSpectatingPlayer.Character:FindFirstChild("Humanoid") then
            Camera.CameraSubject = activeSpectatingPlayer.Character.Humanoid
            if activeSpectatingPlayer.Character:FindFirstChild("HumanoidRootPart") then
                LocalPlayer:RequestStreamAroundAsync(activeSpectatingPlayer.Character.HumanoidRootPart.Position)
            end
        else
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
                Camera.CameraSubject = LocalPlayer.Character.Humanoid
            end
        end
    elseif activeSpectatingModel then
        if activeSpectatingModel and activeSpectatingModel:FindFirstChild("Humanoid") then
            Camera.CameraSubject = activeSpectatingModel.Humanoid
            if activeSpectatingModel:FindFirstChild("HumanoidRootPart") then
                LocalPlayer:RequestStreamAroundAsync(activeSpectatingModel.HumanoidRootPart.Position)
            end
        else
            if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
                Camera.CameraSubject = LocalPlayer.Character.Humanoid
            end
        end
    else
        if LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("Humanoid") then
            if Camera.CameraSubject ~= LocalPlayer.Character.Humanoid then
                Camera.CameraSubject = LocalPlayer.Character.Humanoid
            end
        end
    end

    if scanningEnabled then
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj:IsA("Model") and obj:FindFirstChild("Humanoid") and obj:FindFirstChild("HumanoidRootPart") then
                local isPlayerChar = false
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character == obj then isPlayerChar = true break end
                end
                if not isPlayerChar and not espCache[obj] then
                    addESP(obj, obj.Name)
                end
            end
        end

        for char, data in pairs(espCache) do
            if data.rootPart and data.textLabel and data.rootPart.Parent then
                local pos = data.rootPart.Position
                data.textLabel.Text = string.format("[%s]\nPos: X:%.0f, Y:%.0f, Z:%.0f", data.name, pos.X, pos.Y, pos.Z)
            else
                removeESP(char)
            end
        end
    end
end)

local function updateActiveBarColor()
    for _, itemFrame in pairs(ScrollingFrame:GetChildren()) do
        if itemFrame:IsA("Frame") then
            local btn = itemFrame:FindFirstChild("PlayerButton")
            if btn then
                if not isBotMode and activeSpectatingPlayer and itemFrame.Name == activeSpectatingPlayer.Name then
                    btn.BackgroundColor3 = Color3.fromRGB(255, 215, 0)
                    btn.TextColor3 = Color3.fromRGB(0, 0, 0)
                elseif isBotMode and activeSpectatingModel and itemFrame.Name == activeSpectatingModel.Name then
                    btn.BackgroundColor3 = Color3.fromRGB(255, 215, 0)
                    btn.TextColor3 = Color3.fromRGB(0, 0, 0)
                else
                    btn.BackgroundColor3 = Color3.fromRGB(30, 30, 35)
                    btn.TextColor3 = Color3.fromRGB(200, 200, 200)
                end
            end
        end
    end
end

local function teleportToPlayerFront(targetPlayer)
    local myChar = LocalPlayer.Character
    local targetChar = targetPlayer.Character
    
    if myChar and myChar:FindFirstChild("HumanoidRootPart") and targetChar and targetChar:FindFirstChild("HumanoidRootPart") then
        local targetHRP = targetChar.HumanoidRootPart
        local myHRP = myChar.HumanoidRootPart
        
        local frontCFrame = targetHRP.CFrame * CFrame.new(0, 0, -3.5) * CFrame.Angles(0, math.pi, 0)
        myHRP.CFrame = frontCFrame
    end
end

local function teleportToModelFront(targetModel)
    local myChar = LocalPlayer.Character
    if myChar and myChar:FindFirstChild("HumanoidRootPart") and targetModel and targetModel:FindFirstChild("HumanoidRootPart") then
        local targetHRP = targetModel.HumanoidRootPart
        local myHRP = myChar.HumanoidRootPart
        
        local frontCFrame = targetHRP.CFrame * CFrame.new(0, 0, -3.5) * CFrame.Angles(0, math.pi, 0)
        myHRP.CFrame = frontCFrame
    end
end

Players.PlayerRemoving:Connect(function(player)
    if activeSpectatingPlayer == player then
        activeSpectatingPlayer = nil

    end

end)

local pressStartTime = 0
local isHolding = false

ResetButton.MouseButton1Down:Connect(function()
    pressStartTime = tick()
    isHolding = true
    
    task.spawn(function()
        while isHolding do
            if tick() - pressStartTime >= 2 then
                isBotMode = not isBotMode
                if isBotMode then
                    Title.Text = "BOT SPECTATE"
                    ResetButton.TextColor3 = Color3.fromRGB(255, 50, 50)
                else
                    Title.Text = "PLAYER SPECTATE"
                    ResetButton.TextColor3 = Color3.fromRGB(255, 255, 255)
                end
                activeSpectatingPlayer = nil
                activeSpectatingModel = nil
                updatePlayerList()
                isHolding = false
                break
            end
            task.wait(0.1)
        end
    end)
end)

ResetButton.MouseButton1Up:Connect(function()
    if isHolding then
        isHolding = false
        local holdDuration = tick() - pressStartTime
        if holdDuration < 2 then
            activeSpectatingPlayer = nil
            activeSpectatingModel = nil
            if isBotMode then
                Title.Text = "BOT SPECTATE"
            else
                Title.Text = "PLAYER SPECTATE"
            end
            updateActiveBarColor()
        end
    end
end)
